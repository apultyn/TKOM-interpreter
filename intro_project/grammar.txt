TKOM gramatyka

program 		= { statement } ;
statement 		= ( assigment, ";" )
				| ( expression, ";" )
				| func_definition
				| if_statement
				| while_loop
				| for_loop
				| return_statement
				| block ;

assigment 		= identifier, ( "=" | "+=" | "-=" ), expression ;
func_definition = identifier, "=", "function", "(", identifier_list, ")", block ;
if_statement 	= "if", "(", expression, ")", block, { else_if }, [ "else", block ] ;
else_if 		= "else", "if", "(", expression, ")", block ;

while_loop 		= "while", "(", expression, ")", block ;
for_loop 		= "for", identifier, "in", identifier, block ;
return_statement= "return", expression ";" ;
block 			= "{", { statement }, "}" ;

expression		= logic_or ;
logic_or		= logic_and, { "or", logic_and } ;
logic_and		= equality, { "and", equality } ;
equality		= comparison, { ("==" | "!="), comparison } ;
comparison		= additive, { (">" | ">=" | "<" | "<="), additive } ;
additive		= multiplicative, { ("+" | "-"), multiplicative } ;
multiplicative	= unary, { ( "*" | "/" ), unary } ;
unary			= ("!" | "-"), unary | postfix ;
postfix			= primary, { ( ".", identifier ) | ( "(", argument_list, ")" ) } ;

primary			= integer
				| float
				| string
				| "True" | "False"
				| identifier
				| list_literal
				| dict_literal
				| linq_query
				| ( "(", expression, ")" ) ;

integer 		= "0" | ( digit-positive , { digit } ) ;
float			= integer, ".", integer ;
string			= '"', { unicode_symbol }, '"' ;
identifier		= letter, { alphanumeric } ;
identifier_list = identifier, { ",", identifier } ;
argument_list	= [ expression, { ",", expression } ] ;
list_literal	= "[", [ expression { ",", expression } ] "]" ;
dict_literal	= "{", [ item_literal, { item_literal } ], "}" ;
item_literal	= "(", string, ":", expression, ")" ;

linq_query		= "from", identifier, "in", identifier,
					"select", expression,
					[ "where", expression ],
					[ "order", "by", expression, [ "descending" ] ] ;

unicode_symbol 	= { *symbole UNICODE* } ;
letter			= { *litery* } ;
alphanumeric	= letter | digit | "_" ;



comment-endline = "//", { unicode_symbol }, "\n";
comment-inline 	= "/*", { unicode_symbol }, "*/";


digit-positive 	= "1" | "2" | "3" | "4" | "5" | "6" | "7" | "8" | "9";
digit 			= "0" | digit-positive;
